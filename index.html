using UnityEngine; public class GameManager : MonoBehaviour { [Header("1. Map Generator Settings")] public GameObject treePrefab; // Ped ka 3D Model public GameObject housePrefab; // Ghar ka 3D Model public int totalTrees = 100000; // 1 Lakh Ped public int totalHouses = 1000; // 1000 Ghar public Vector2 terrainSize = new Vector2(5000, 5000); // Map Area [Header("2. Referral & Earn Settings")] public int userWallet = 0; public string myReferralCode = "USER123"; [Header("3. Car Controller Settings")] public GameObject carObject; // Aapki Gadi ka 3D Model public float moveSpeed = 20f; // Aage/Peeche ki Speed public float turnSpeed = 50f; // Left/Right Murne ki Speed void Start() { // Game start hote hi Map banega GenerateForest(); GenerateHouses(); } void Update() { // Har frame mein Gadi Controls check honge HandleCarMovement(); } // --- 1. JUNGLE AUR GHAR GENERATOR --- void GenerateForest() { for (int i = 0; i < totalTrees; i++) { float x = Random.Range(0, terrainSize.x); float z = Random.Range(0, terrainSize.y); Instantiate(treePrefab, new Vector3(x, 0, z), Quaternion.identity); } Debug.Log("1 Lakh Ped Generate Ho Gaye!"); } void GenerateHouses() { for (int i = 0; i < totalHouses; i++) { float x = Random.Range(0, terrainSize.x); float z = Random.Range(0, terrainSize.y); Instantiate(housePrefab, new Vector3(x, 0, z), Quaternion.identity); } Debug.Log("1000 Ghar Generate Ho Gaye!"); } // --- 2. GADI CHALANEY KA LOGIC (Aage, Peeche, Left, Right) --- void HandleCarMovement() { if (carObject == null) return; // W/S ya Up/Down Arrow keys (Aage / Peeche ke liye) float moveInput = Input.GetAxis("Vertical"); // A/D ya Left/Right Arrow keys (Left / Right ke liye) float turnInput = Input.GetAxis("Horizontal"); // Gadi ko Aage/Peeche chalana carObject.transform.Translate(Vector3.forward * moveInput * moveSpeed * Time.deltaTime); // Gadi ko Left/Right Modna (Rotate karna) carObject.transform.Rotate(Vector3.up * turnInput * turnSpeed * Time.deltaTime); } // --- 3. REFERRAL & EARN LOGIC --- public void RedeemReferralCode(string inputCode) { if (inputCode == "FRIEND50") { userWallet += 50; // Wallet mein ₹50 add honge Debug.Log("Referral Successful! Naya Balance: ₹" + userWallet); } else { Debug.Log("Invalid Referral Code!"); } } }
